# The image viewer (apps/images; M9 step D): each format, fitted; next and previous; zoom and moving about; Open
# (C10): the list typed to BIG.PNG (too big for a thumbnail: described), then down to D.GIF (its
# thumbnail), opened. (The list's Modified column is blanked in the shots: the dates are the copy's.)
run IMAGES.EXE C:\GUI\IMG\A.PNG
wait "A.PNG - Images"
shot png "A.PNG - Images"
key pgdn
wait "B.JPG - Images"
shot jpg "B.JPG - Images"
key pgdn
wait "BIG.PNG - Images"
shot big-fit "BIG.PNG - Images"
key 1
shot big-actual "BIG.PNG - Images"
key right right down
shot big-moved "BIG.PNG - Images"
key - -
shot big-small "BIG.PNG - Images"
key f pgdn
wait "C.BMP - Images"
shot bmp "C.BMP - Images"
key pgdn
wait "D.GIF - Images"
shot gif "D.GIF - Images"
key pgdn
wait "E.TGA - Images"
shot tga "E.TGA - Images"
key pgdn
wait "F.PPM - Images"
shot ppm "F.PPM - Images"
key pgdn
wait "G.PCX - Images"
shot pcx8 "G.PCX - Images"
key pgdn
wait "H.PCX - Images"
shot pcx24 "H.PCX - Images"
key pgdn
wait "A.PNG - Images"
key pgup
wait "H.PCX - Images"
key ctrl+o
wait "Open"
key shift+tab
type big
shot open-big "Open" 355 56 110 121
key down down
shot open-gif "Open" 355 56 110 121
key enter
gone "Open"
wait "D.GIF - Images"
key alt+f4
gone "D.GIF - Images"
expect IMAGES open C:\GUI\IMG\B.JPG 160x120 JPG
expect IMAGES open C:\GUI\IMG\G.PCX 160x120 PCX
expect IMAGES open C:\GUI\IMG\H.PCX 160x120 PCX
expect IMAGES open C:\GUI\IMG\D.GIF 160x120 GIF
expect IMAGES end
