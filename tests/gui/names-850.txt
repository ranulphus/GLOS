# DOS file names through code page 850 (M10 step G; GUITEST names, GLOSCP from GUITEST's profile): files named
# with 850's letters, shown by the file dialog as text; the one chosen comes back as DOS's bytes.
config program:GUITEST.EXE env GLOSCP=850
run GUITEST.EXE names C:\GUI\CP850
wait "Open"
shot names "Open"
key shift+tab down down
shot chosen "Open"
key enter
gone "Open"
expect GUI page 850
expect GUI chose C:\GUI\CP850\\x8FRHUS.TXT
expect GUI end
