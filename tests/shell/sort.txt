# letters in any case, and with accents, as one; a line that begins another first; /R the other way
> sort s2.txt
> sort /r s2.txt
> sort < s2.txt > s.txt
cat C:\S.TXT
> type s2.txt | sort /R
> echo %errorlevel%
# equal lines: in the order they came, /R too (MS-DOS 7.1's: as its sort leaves them, "a A a B B b b")
> sort stab.txt
> sort /r stab.txt
# /+n: from the n-th character; a shorter line has nothing there and goes first; the last /+ counts
> sort /+3 short.txt
> sort /+2 short.txt
> sort /r /+2 short.txt
> sort /+5 short.txt
> sort /+70000 short.txt
> sort /+2 /+3 short.txt
> sort /+2/R short.txt
# code page 437's order, MS-DOS 7.1's: case and accents aside, cents, pounds and yen with $, sharp s with S
> sort acc.txt
> sort col.txt
# line ends: LF alone, a CR before LF the line's own, none at the end (none written, if last), Ctrl+Z
> sort slf.txt > s.txt
cat C:\S.TXT
> sort scr.txt > s.txt
cat C:\S.TXT
> sort noeol.txt > s.txt
cat C:\S.TXT
