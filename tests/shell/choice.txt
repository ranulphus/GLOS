# CHOICE and SET /P (sdk/shell, 2026-11-25: the setup floppy's installer, and DOS batch files natively): CHOICE
# as MS-DOS 6's (ERRORLEVEL the key's place, for IF ERRORLEVEL too); SET /P a line typed.
> choice
key n
> echo %errorlevel%
> if errorlevel 2 echo two or more
> choice /c:abc Pick one
key x
key B
> echo %errorlevel%
> if errorlevel 3 echo three
> if errorlevel 2 echo two
> choice /c:yn /s Case counts
key Y
key y
> echo %errorlevel%
> choice /n /c:12 "Quiet:"
key 2
> echo %errorlevel%
> choice /c:yn Ctrl+C
key CTRLC
> echo %errorlevel%
nokeys
> choice /c:ynq /t:q,5 Nobody here
> echo %errorlevel%
> choice /c:ab Nobody, no default
> echo %errorlevel%
keys
# a wrong key, then nobody to press another (a job's input ended): /T's answer, not the first key
> choice /c:yn /t:n,5 Install now
key x
nokeys
key z
> echo %errorlevel%
keys
# what it refuses
> choice /c:yn /t:x,5 Wrong default
> echo %errorlevel%
> choice /z
> choice /t:y
