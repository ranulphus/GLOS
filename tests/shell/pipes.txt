# a pipe: the first's output to a file, the second reading it; the file gone after
> a | b
ended 0
ended 5
> echo %ERRORLEVEL%
> dir C:\ /b
# three steps, and what follows the pipe goes by its last program
> a x | b y | a z && echo worked

ended 0
ended 0
> a | b || echo the last one failed
ended 0
