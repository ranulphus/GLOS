# asks: not for the names, not "Compare more files (Y/N)?". Exit codes: 0 the same, 1 different, 2 no file.

# the same; bytes that differ (hex, /D decimal, /A characters, /L lines); /C case aside
> comp a.txt b.txt
> echo %errorlevel%
> comp a.txt c.txt
> echo %errorlevel%
> comp a.txt c.txt /d
> comp a.txt c.txt /a
> comp a.txt c.txt /l
> comp a.txt d.txt /c
> comp a.txt c.txt /c /a
# ten at most
> comp g.txt h.txt
# sizes: not compared, but with /N=n (the first n lines)
> comp a.txt e.txt
> echo %errorlevel%
> comp a.txt e.txt /n=1
> comp a.txt f.txt /n=1
> comp a.txt f.txt /n=2 /l
# a folder second: the same name in it; patterns
> comp w\a.txt v
> comp w\*.txt v\
> comp w\*.txt v\*.bak
> comp w\*.txt v\a.txt
# what isn't there, and what can't be run
> comp a.txt nosuch.txt
> echo %errorlevel%
> comp nosuch*.txt a.txt
> comp a.txt
> echo %errorlevel%
> comp
> comp a.txt b.txt c.txt
> comp /x a.txt b.txt
> exit
