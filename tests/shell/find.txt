# each file under its heading, after an empty line; /I, /V, /C, /N
> find "hello" a.txt
> find /i "hello" a.txt
> find /v "hello" a.txt
> find /c "hello" a.txt
> find /n "hello" a.txt
> find /n /v /i "hello" a.txt
> find /c /v "hello" a.txt
> find /c /n "hello" a.txt
> find "hello" a.txt b.txt c.txt
> echo %errorlevel%
> find "hello" a.txt > f.txt
cat C:\F.TXT
> find /c "hello" a.txt b.txt > f.txt
cat C:\F.TXT
# the input (a pipe, or <): no heading
> find "hello" < a.txt
> type a.txt | find /c "hello"
> find /n "hello" < a.txt
> find /c /v "hello" < a.txt > f.txt
cat C:\F.TXT
> find "zzz" < nul
> find /c "zzz" < nul
> find /c "zzz" < empty.txt
# names as typed; switches anywhere; spaces, commas and semicolons between words
> find /c "hello" c:\sub\d.txt
> find "hello" .\a.txt /c
> find "hello" /C a.txt,b.txt;c.txt
> find "hello"a.txt
> find /V /N "hello"	c.txt

> find "" a.txt
> echo %errorlevel%
> find /v "" a.txt
